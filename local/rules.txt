# 결정적 검사 규칙 (LLM 응답에서 위반 항목은 자동 제거)
# KEEP CONFIG_X=값 : 반드시 이 값 유지
# OFF  CONFIG_X    : 반드시 꺼짐 (켜는 제안 제거)
# BAN  CONFIG_X=값 : 이 값은 쓰지 않음 (확인된 실패)

# --- 반드시 유지 ---
KEEP CONFIG_64BIT=y
KEEP CONFIG_RELOCATABLE=y
KEEP CONFIG_ACPI=y            # x86에서 끄면 부팅 불안정
KEEP CONFIG_PCI=y             # ACPI 동작에 필요
KEEP CONFIG_BLK_DEV_INITRD=y
KEEP CONFIG_RD_LZ4=y
KEEP CONFIG_KERNEL_LZ4=y
KEEP CONFIG_SYSFB=y
KEEP CONFIG_SYSFB_SIMPLEFB=y
KEEP CONFIG_FB=y
KEEP CONFIG_FB_DEVICE=y
KEEP CONFIG_FB_SIMPLE=y
KEEP CONFIG_DEVTMPFS=y
KEEP CONFIG_PROC_FS=y
KEEP CONFIG_SYSFS=y
KEEP CONFIG_TMPFS=y
KEEP CONFIG_BINFMT_ELF=y
KEEP CONFIG_TTY=y
KEEP CONFIG_SERIAL_8250=y
KEEP CONFIG_SERIAL_8250_CONSOLE=y
KEEP CONFIG_PRINTK=y
KEEP CONFIG_PRINTK_TIME=y
KEEP CONFIG_MODULES=y         # 이후 드라이버 점진 복원 계획

# --- 반드시 끔 ---
OFF CONFIG_DRM                # 디스플레이 재초기화
OFF CONFIG_VT                 # 계기판 화면을 덮어씀
OFF CONFIG_FRAMEBUFFER_CONSOLE
OFF CONFIG_FB_VESA            # simplefb와 충돌
OFF CONFIG_FB_EFI

# --- 확인된 실패 ---
BAN CONFIG_SERIAL_8250_NR_UARTS=1          # ttyS0이 0x2f8에 붙어 시리얼 출력이 사라짐
BAN CONFIG_SERIAL_8250_RUNTIME_UARTS=1
